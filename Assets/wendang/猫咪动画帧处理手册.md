## 处理流程

### 第1步：放视频

将视频放入 `Assets\Videos\Cat\{猫咪名}\`，文件名格式：`{猫咪名}_{表情名}_{序号}.mp4`

示例：`黑炭_smug_00001_.mp4`

### 第2步：FFmpeg 切帧 + 抽帧

```powershell
$ffmpeg = "C:\ffmpeg\bin\ffmpeg.exe"
$video = "视频路径"
$framesDir = "输出目录\frames"

# 切帧：每秒12帧
& $ffmpeg -i $video -vf "fps=12" "$framesDir\frame_%04d.png" -y

# 抽帧：每隔1帧取1帧（62帧→31帧）
$frames = Get-ChildItem $framesDir -Filter "*.png" | Sort-Object Name
$count = 0
for ($i = 0; $i -lt $frames.Count; $i += 2) {
    Copy-Item $frames[$i].FullName "输出目录\selected\frame_$('{0:D4}' -f $count).png"
    $count++
}
```

### 第3步：Rembg AI 抠图

```python
import os
os.environ["NUMBA_DISABLE_JIT"] = "1"

import gc
from rembg import remove, new_session
from PIL import Image

session = new_session("bria-rmbg")

for f in sorted(os.listdir("selected目录")):
    img = Image.open(f)
    if max(img.size) > 512:
        img.thumbnail((512, 512), Image.Resampling.LANCZOS)
    result = remove(img, session=session)
    result.save(f"输出目录/{f}")
    gc.collect()
```

### 第4步：复制到资源目录

```powershell
Copy-Item "输出目录\cutout\*.png" "Assets\Art\Cat\Frames\{猫咪名}\{表情名}\" -Force
```

## 一键脚本

将以下脚本保存为 `process_cat_video.py`，用法：

```bash
python process_cat_video.py <视频路径> <猫咪名> <表情名>
```

```python
import os
os.environ["NUMBA_DISABLE_JIT"] = "1"

import sys
import subprocess
import shutil
import gc
from rembg import remove, new_session
from PIL import Image

FFMPEG = r"C:\ffmpeg\bin\ffmpeg.exe"
PROJECT = r"E:\UnityProject\wuziqi"

def process(video_path, cat_name, expression):
    # 临时目录
    tmp = os.path.join(PROJECT, f"_tmp_{cat_name}_{expression}")
    frames_dir = os.path.join(tmp, "frames")
    select_dir = os.path.join(tmp, "selected")
    cutout_dir = os.path.join(tmp, "cutout")
    os.makedirs(frames_dir, exist_ok=True)
    os.makedirs(select_dir, exist_ok=True)
    os.makedirs(cutout_dir, exist_ok=True)

    # 1. 切帧
    print("[1/4] FFmpeg extracting frames...")
    subprocess.run([FFMPEG, "-i", video_path, "-vf", "fps=12",
                    os.path.join(frames_dir, "frame_%04d.png"), "-y"],
                   capture_output=True)

    # 2. 抽帧
    print("[2/4] Selecting frames...")
    files = sorted([f for f in os.listdir(frames_dir) if f.endswith(".png")])
    count = 0
    for i in range(0, len(files), 2):
        src = os.path.join(frames_dir, files[i])
        dst = os.path.join(select_dir, f"frame_{count:04d}.png")
        shutil.copy2(src, dst)
        count += 1
    print(f"  Selected {count} frames")

    # 3. 抠图
    print("[3/4] Removing background (bria-rmbg)...")
    session = new_session("bria-rmbg")
    selected = sorted([f for f in os.listdir(select_dir) if f.endswith(".png")])
    for i, f in enumerate(selected):
        img = Image.open(os.path.join(select_dir, f))
        if max(img.size) > 512:
            img.thumbnail((512, 512), Image.Resampling.LANCZOS)
        result = remove(img, session=session)
        result.save(os.path.join(cutout_dir, f))
        gc.collect()
        print(f"  [{i+1}/{len(selected)}] {f}")

    # 4. 复制到资源目录
    print("[4/4] Copying to assets...")
    target = os.path.join(PROJECT, "Assets", "Art", "Cat", "Frames", cat_name, expression)
    os.makedirs(target, exist_ok=True)
    for f in os.listdir(cutout_dir):
        shutil.copy2(os.path.join(cutout_dir, f), os.path.join(target, f))

    # 清理
    shutil.rmtree(tmp)
    print(f"Done! Output: {target}")
    print(f"Frames: {count}")

if __name__ == "__main__":
    if len(sys.argv) < 4:
        print("Usage: python process_cat_video.py <video> <cat_name> <expression>")
        print('Example: python process_cat_video.py "黑炭_smug.mp4" 黑炭 smug')
        sys.exit(1)
    process(sys.argv[1], sys.argv[2], sys.argv[3])
```

## 注意事项

1. **模型首次下载**：bria-rmbg 模型约 977MB，首次会自动下载到 C:\Users\zhn\.rembg\models\
2. **内存限制**：如果报内存不足，用 img.thumbnail((512, 512)) 缩小图片
3. **网络问题**：下载模型需要翻墙或用国内镜像（gh-proxy.com）
4. **帧数**：一般 31 帧足够，从 62 帧中每隔 1 帧抽取
5. **NUMBA 缓存问题**：设置 NUMBA_DISABLE_JIT=1 可避免报错
